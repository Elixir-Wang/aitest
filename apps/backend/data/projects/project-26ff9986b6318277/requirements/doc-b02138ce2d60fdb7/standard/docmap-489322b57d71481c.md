# 通用模型网关设计

> 提供统一接入 LLM 功能平台（三方，公司内）

## 需求背景

现在模型网关服务 agent-chat-model-g 服务与业务有较大的耦合和绑定，不适合作为统一的网关层使用，故而将核心的与大模型 LLM 交互功能提取出来成为通用功能沉淀。

- **llm-gateway-g**：web 服务应用，可提供给其他业务方使用 OPEN AI 的通信协议，调用各种大模型。
- **llm-gateway-g-sdk**：核心调用 LLM 的能力，提供 go sdk 给 go 相关服务使用，减少与 llm-gateway-g 通过 HTTP 交互带来的性能损耗。

## 数据库表设计

```sql
SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;

-- ----------------------------
-- Table structure for api_keys
-- ----------------------------
DROP TABLE IF EXISTS `api_keys`;
CREATE TABLE `api_keys` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `api_key` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `source` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `creat_time` datetime(3) DEFAULT NULL,
  `is_delete` int NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`) USING BTREE,
  UNIQUE KEY `idx_api_keys_api_key` (`api_key`) USING BTREE,
  KEY `idx_api_keys_source_delete` (`source`,`is_delete`) USING BTREE,
  KEY `idx_service_api_keys_source_delete` (`source`,`is_delete`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci ROW_FORMAT=DYNAMIC;

-- ----------------------------
-- Table structure for llm_info_classify
-- ----------------------------
DROP TABLE IF EXISTS `llm_info_classify`;

  `id` bigint unsigned NOT NULL AUTO_INCREMENT COMMENT 'Primary key',
  `classify_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `user_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `icon` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci DEFAULT NULL,
  `create_time` datetime(3) DEFAULT NULL,
  `update_time` datetime(3) DEFAULT NULL,
  `is_delete` bigint DEFAULT '0',
  PRIMARY KEY (`id`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci ROW_FORMAT=DYNAMIC COMMENT='LLM vendor/classification';

-- Table structure for llm_model_config
-- ----------------------------
DROP TABLE IF EXISTS `llm_model_config`;

  `id` bigint NOT NULL AUTO_INCREMENT,
  `llm_code` char(6) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,

  `end_point` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `api_key` text CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,

  `proxy_id` bigint DEFAULT NULL,
  `classify_id` bigint DEFAULT NULL,
  `create_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `update_name` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL,
  `create_time` datetime(3) DEFAULT NULL,
  `update_time` datetime(3) DEFAULT NULL,
  `is_delete` int NOT NULL DEFAULT '0',
  `protocol_options` json DEFAULT NULL,
  PRIMARY KEY (`id`) USING BTREE,
  KEY `uniq_index_llm_code` (`llm_code`) USING BTREE,
  KEY `idx_llm_name` (`llm_name`) USING BTREE,
  KEY `idx_target_model` (`target_model`) USING BTREE,

  KEY `idx_llm_model_config_name_delete` (`is_delete`) USING BTREE

-- Table structure for proxy_config
-- ----------------------------

-- ----------------------------

  KEY `idx_source_api_key_delete` (`source_api_key`,`is_delete`) USING BTREE

SET FOREIGN_KEY_CHECKS = 1;
```
